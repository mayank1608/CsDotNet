1. Create solution for project
mkdir CarRentalSystem
cd CarRentalSystem
dotnet new sln -n CarRentalSystem 

2. Create library for CarRentalBL(Business laryer)
dotnet new classlib -n CarRentalBL

3. Create library for CarRentalDTO(Data transfer object)
dotnet new classlib -n CarRentalDTO

3. Create console app for CarRentalPL(Presentation layer)
dotnet new console -n CarRentalPL

4. Add solution for CarRentalBL & CarRentalPL & CarRentalDTO
dotnet sln add CarRentalBL/CarRentalBL.csproj
dotnet sln add CarRentalPL/CarRentalPL.csproj
dotnet sln add CarRentalDTO/CarRentalDTO.csproj

5. Add reference of CarRentalBL in CarRentalPL
dotnet add CarRentalPL/CarRentalPL.csproj reference CarRentalBL/CarRentalBL.csproj

6. Add reference of CarRentalDTO in CarRentalPL
dotnet add CarRentalPL/CarRentalPL.csproj reference CarRentalDTO/CarRentalDTO.csproj

7. Add reference of CarRentalDTO in CarRentalBL
dotnet add CarRentalBL/CarRentalBL.csproj reference CarRentalDTO/CarRentalDTO.csproj

// auto generate controller using model
// tool:
dotnet tool update --global dotnet-aspnet-codegenerator
// package:
dotnet add package Microsoft.VisualStudio.Web.CodeGeneration.Design
//run cmd after above cmds
dotnet restore

Assuming:
Model = Product
DbContext = AppDbContext

dotnet aspnet-codegenerator controller \
  -name ProductsController \
  -async \
  -api \
  -m Product \
  -dc AppDbContext \
  -outDir Controllers
  
// Single line
 dotnet aspnet-codegenerator controller -name ProductsController -async -api -m Product -dc AppDbContext -outDir Controllers
